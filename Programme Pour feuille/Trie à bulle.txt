void tri_a_bulle(int *tab, int taille) {
    int echange = 1;
    int temp;
    int limite = taille - 1;
    
    while (echange) {
        echange = 0;
        for (int i = 0; i < limite; i++) {
            if (tab[i] > tab[i + 1]) {
                temp = tab[i];
                tab[i] = tab[i + 1];
                tab[i + 1] = temp;
                echange = 1;
            }
        }
        limite--;
    }
}
